  const failed = results.filter((r) => !r.ok);
  markerEl.textContent = 'HARNESS_RESULT total=' + results.length + ' failed=' + failed.length + ' ' +

}
function report(name, ok, detail) {
  results.push({ name, ok, detail: detail === undefined ? null : detail });
  log.push((ok ? 'PASS ' : 'FAIL ') + name +
    (detail !== undefined && detail !== null ? '  ::  ' + JSON.stringify(detail) : ''));
  render();
}

/** Cheap order-sensitive checksum so we can tell whether a grid changed. */
function checksum(arr) {
  let h = 2166136261 >>> 0;
  for (let i = 0; i < arr.length; i += 7) {
    h ^= Math.round(arr[i]) & 0xff;
    h = Math.imul(h, 16777619) >>> 0;
  }
  return h.toString(16);
}

function scanFinite(w) {
  let bad = 0;
  for (const key of ['T', 'albedo', 'precip', 'vegetation', 'habitability', 'ice']) {
    const arr = w.grid[key];
    if (!arr) continue;
    for (let i = 0; i < arr.length; i++) if (!isFinite(arr[i])) bad++;
  }
  return bad;
}

          gwx.GX === 180 && gwx.GY === 90, { gx: gwx.GX, gy: gwx.GY });
      }
    }
    doc.getElementById('wizCreate').click();
    await sleep(1200);
    const nw = api.world();
    report('a planet can be created at a custom resolution',
      nw.terrain.GX === 180 && nw.grid.NB === 24,
      { terrain: nw.terrain.GX + 'x' + nw.terrain.GY, climate: nw.grid.NB + 'x' + nw.grid.NL });
    report('custom-resolution planet simulates normally', (() => {
      for (let i = 0; i < 12; i++) nw.step(1);
      return isFinite(nw.tGlobal) && nw.tGlobal > 200 && nw.tGlobal < 340;
    })(), { tGlobal: +nw.tGlobal.toFixed(2) });

    // preset picker asks for a resolution before building
    doc.getElementById('presetSel').value = 'mars';
    doc.getElementById('presetSel').dispatchEvent(new cw.Event('change', { bubbles: true }));
    await sleep(300);
    const gridModal = doc.getElementById('gridModal');
    report('choosing a preset asks for the mesh count first',
      !gridModal.classList.contains('hidden'), { visible: !gridModal.classList.contains('hidden') });
    const countNum = doc.getElementById('gridCountNum');
    countNum.value = '16200';
    countNum.dispatchEvent(new cw.Event('change', { bubbles: true }));
    await sleep(150);
    doc.getElementById('gridConfirm').click();
    await sleep(1500);
    const marsWorld = api.world();
    report('the preset is built at the chosen resolution',
      marsWorld.terrain.GX === 180 && Math.round(api.state.params.irradiance) === 586,
      { terrain: marsWorld.terrain.GX + 'x' + marsWorld.terrain.GY, irradiance: Math.round(api.state.params.irradiance) });
    api.applyPreset('earth');
    await sleep(900);
  } catch (e) { gridErr = (e && e.stack ? e.stack.split('\n').slice(0, 2).join(' | ') : e.message); }
  report('grid resolution flow without errors', gridErr === null, gridErr);

  /* ---------------- physics invariants ---------------- */
  const { World } = await import('/src/world/world.js');
  const { defaults } = await import('/src/state/params.js');
  const { appDefaultParams } = await import('/src/state/derive.js');
  const { transportAirStep } = await import('/src/physics/climate.js');
  // appDefaultParams runs the very same applyAutoRules the app runs at startup, so
  // these numbers are the ones the panel shows. Re-implementing the rules by hand is
  // exactly what let a "calibrated in Node, snowballing in the browser" mismatch
  // survive: the hand-rolled version fed defaults().humidity (68) into the cloud rule,
  // modelling cloud cover 60.96 while the app actually ran at 68.2.
  const earthParams = () => appDefaultParams(defaults);

  const pw = new World(earthParams(), { seed: defaults().seed });
  const annualMarks = [];
  for (let yr = 1; yr <= 60; yr++) {
    for (let i = 0; i < 12; i++) pw.step(1);
    if (yr === 1 || yr === 40 || yr === 60) annualMarks.push({ yr, t: pw.tGlobalAnnual });
  }
  const yr1 = annualMarks[0].t, yr40 = annualMarks[1].t, yr60 = annualMarks[2].t;
  report('50-year integration stays bounded', isFinite(pw.tGlobal) && pw.tGlobal > 200 && pw.tGlobal < 340,
    +pw.tGlobal.toFixed(2));
  report('Earth default settles at an Earth-like 12-month mean',
    yr60 > 285 && yr60 < 292, +yr60.toFixed(2));
  report('year 1 is already near equilibrium, not plunging',
    yr1 > 284 && Math.abs(yr1 - yr60) < 2.5, { year1: +yr1.toFixed(2), year60: +yr60.toFixed(2) });
  report('Earth default is a fixed point, not a slow slide',
    Math.abs(yr60 - yr40) < 0.25, { year40: +yr40.toFixed(3), year60: +yr60.toFixed(3) });

  // The root cause of the old "steady slide into the snowball": transportAirStep
  // conserved the unweighted band sum ΣT instead of Σ(cosφ·T), so every poleward
  // exchange quietly dropped heat. The conserved quantity must be the one the
  // global averages use.
  {
    const g = pw.grid;
    const weighted = () => {
      let acc = 0;
      for (let j = 0; j < g.NB; j++) acc += Math.cos((g.lats[j] * Math.PI) / 180) * g.airT[j];
      return acc;
    };
    for (let j = 0; j < g.NB; j++) {
      g.airT[j] = 320 - 60 * Math.pow(Math.abs(g.lats[j]) / 90, 2);
    }
    const before = weighted();
    transportAirStep(g, pw.params, 2.63e6);
    transportAirStep(g, pw.params, 2.63e6);
    const drift = Math.abs(weighted() - before) / Math.max(1, Math.abs(before));
    report('meridional transport conserves area-weighted heat', drift < 1e-9, drift.toExponential(2));
  }

  let poleAcc = 0;
  for (let i = 0; i < pw.grid.NL; i++) poleAcc += pw.grid.T[i];
  const poleT = poleAcc / pw.grid.NL;
  report('polar temperature is physical', poleT > 180 && poleT < 300, +poleT.toFixed(1));
  report('no non-finite values in climate fields', scanFinite(pw) === 0, { bad: scanFinite(pw) });

  /* ---------------- CO2 sensitivity ---------------- */
  const sensitivity = [];
  let sensLo = Infinity, sensHi = -Infinity;
  for (const co2 of [280, 420, 560, 840]) {
    const w2 = new World(Object.assign(earthParams(), { co2 }), { seed: 4242 });
    for (let i = 0; i < 480; i++) w2.step(1);
    sensitivity.push(co2 + 'ppm=' + w2.tGlobalAnnual.toFixed(2));
    sensLo = Math.min(sensLo, w2.tGlobalAnnual);
    sensHi = Math.max(sensHi, w2.tGlobalAnnual);
  }
  report('climate responds to CO2 across the range', sensHi > sensLo + 2, sensitivity);

  finish();
}

function finish() {
  const failed = results.filter((r) => !r.ok);
  log.push('');
  log.push('==== ' + (results.length - failed.length) + '/' + results.length + ' passed, ' + failed.length + ' failed ====');
  render();
  const payload = JSON.stringify({ total: results.length, failed: failed.length, results }, null, 1);
  fetch('/__result', { method: 'POST', body: payload })
    .then(() => { document.title = failed.length ? 'FAILED' : 'ALL PASS'; })
    .catch(() => { document.title = 'POST-FAILED'; });
}

main().catch((e) => { report('harness crashed', false, e.stack || e.message); finish(); });
